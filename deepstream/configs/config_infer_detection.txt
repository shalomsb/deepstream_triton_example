# nvinferserver config for yolo26x_ensemble using the DETECTION postprocess path
# (vs config_infer.txt which uses `postprocess { other {} }` + output_tensor_meta).
#
# This routes detections through attachDetectionMetadata() so nvinferserver sets
# frameMeta->bInferDone -> nvtracker assigns real object_ids. See psm/FINDINGS.md
# ("Option A"). The custom_lib parser splits the ensemble's three output layers.
#
# nvinferserver uses a protobuf schema (NOT the nvinfer INI format). Fields below
# are verified against nvdsinferserver_config.proto + nvdsinferserver_common.proto:
#   PostProcessParams { labelfile_path; oneof: detection|classification|other }
#   DetectionParams   { num_detected_classes; custom_parse_bbox_func; oneof cluster }
#   CustomLib         { path }
# simple_cluster is a THRESHOLD-ONLY filter (no box merging) -> the nvinferserver
# equivalent of nvinfer cluster-mode=4 (None), which is exactly right for our
# already-post-NMS ensemble output. (Nms is marked "not supported yet" here.)

infer_config {
  unique_id: 1
  gpu_ids: [0]
  max_batch_size: 8

  backend {
    triton {
      model_name: "yolo26x_ensemble"
      version: -1
      model_repo {
        root: "/triton/model_repo"
        log_level: 2
        strict_model_config: true
        pinned_memory_pool_byte_size: 1710886400
        cuda_device_memory { device: 0, memory_pool_byte_size: 671088640 }
      }
    }
    output_mem_type: MEMORY_TYPE_CPU
  }

  preprocess {
    network_format: IMAGE_FORMAT_RGB
    tensor_order: TENSOR_ORDER_LINEAR
    normalize {
      scale_factor: 1.0
    }
    maintain_aspect_ratio: 0
    frame_scaling_filter: 1
  }

  postprocess {
    labelfile_path: "/deepstream/labels.txt"
    detection {
      num_detected_classes: 80
      custom_parse_bbox_func: "NvDsInferParseCustomYolo26Ensemble"
      # ensemble already ran NMS; use the lightest clustering as a pass-through.
      simple_cluster {
        threshold: 0.25
      }
    }
  }

  # the compiled parser (built by scripts/build_parser.sh into /deepstream/parser)
  custom_lib {
    path: "/deepstream/parser/libnvds_yolo26_ensemble_parser.so"
  }
}

input_control {
  process_mode: PROCESS_MODE_FULL_FRAME
  interval: 0
}
